cmake_minimum_required(VERSION 3.16)
project(stand_go2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
list(APPEND CMAKE_PREFIX_PATH "/opt/unitree_robotics/lib/cmake" "/usr/local/lib/cmake")
find_package(unitree_sdk2 REQUIRED)
set(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
include_directories("${PROJECT_ROOT}/common")
foreach(app stand_go2 go2_sport_adapter)
  add_executable(${app} ${app}.cpp)
  target_link_libraries(${app} unitree_sdk2)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Boost REQUIRED COMPONENTS program_options)
  foreach(test go2_posture_test go2_session_test go2_service_test go2_config_test go2_rpc_probe go2_state_probe go2_adapter_test)
    add_executable(${test} "${PROJECT_ROOT}/tests/${test}.cpp")
    target_link_libraries(${test} unitree_sdk2)
  endforeach()
  target_compile_definitions(go2_config_test PRIVATE TEST_CONFIG_PATH="${PROJECT_ROOT}/simulate/config.yaml")
  target_include_directories(go2_config_test PRIVATE "${PROJECT_ROOT}/simulate/src")
  target_link_libraries(go2_config_test yaml-cpp Boost::program_options)
  add_executable(go2_bridge_lifecycle_test "${PROJECT_ROOT}/tests/go2_bridge_lifecycle_test.cpp" "${PROJECT_ROOT}/simulate/src/joystick/joystick.cc")
  target_include_directories(go2_bridge_lifecycle_test PRIVATE "${PROJECT_ROOT}/simulate/src" "${PROJECT_ROOT}/simulate/mujoco/include")
  target_compile_definitions(go2_bridge_lifecycle_test PRIVATE TEST_GO2_SCENE="${PROJECT_ROOT}/unitree_robots/go2/scene.xml")
  target_link_libraries(go2_bridge_lifecycle_test unitree_sdk2 yaml-cpp Boost::program_options "${PROJECT_ROOT}/simulate/mujoco/lib/libmujoco.so")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME go2_cli_test COMMAND "${Python3_EXECUTABLE}" "${PROJECT_ROOT}/tests/go2_cli_test.py" "$<TARGET_FILE:stand_go2>")
  foreach(test go2_posture_test go2_session_test go2_service_test go2_config_test go2_bridge_lifecycle_test)
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES TIMEOUT 15)
  endforeach()
endif()
